char a[3072]; char b[3072];
int idx(int x,int y){ return ((y+48)%48)*64 + ((x+64)%64); }
int n8(int x,int y){ int n; n=0;
  n+=a[idx(x-1,y-1)]; n+=a[idx(x,y-1)]; n+=a[idx(x+1,y-1)];
  n+=a[idx(x-1,y)];                      n+=a[idx(x+1,y)];
  n+=a[idx(x-1,y+1)]; n+=a[idx(x,y+1)]; n+=a[idx(x+1,y+1)];
  return n; }
int main(){ int x; int y; int i; int n;
  for(i=0;i<3072;i++) a[i] = (rnd()%5)==0;
  while(1){
    cls();
    for(y=0;y<48;y++) for(x=0;x<64;x++){
      if(a[idx(x,y)]) rect(x*5,y*5,4,4,0xC8FF5A);
      n = n8(x,y);
      b[idx(x,y)] = a[idx(x,y)] ? (n==2||n==3) : (n==3);
    }
    for(i=0;i<3072;i++) a[i]=b[i];
    flip();
  }
  return 0; }
