int sx[256]; int sy[256]; int len; int dx; int dy; int fx; int fy; int t;
int main(){
  int i; int hx; int hy;
  len=4; dx=1; dy=0; sx[0]=8; sy[0]=6; fx=14; fy=9; t=0;
  while(1){
    if(key(37)) { dx=0-1; dy=0; }
    if(key(39)) { dx=1; dy=0; }
    if(key(38)) { dx=0; dy=0-1; }
    if(key(40)) { dx=0; dy=1; }
    t++;
    if(t%6==0){
      hx=sx[0]+dx; hy=sy[0]+dy;
      if(hx<0) hx=31; if(hx>31) hx=0; if(hy<0) hy=23; if(hy>23) hy=0;
      for(i=len;i>0;i--){ sx[i]=sx[i-1]; sy[i]=sy[i-1]; }
      sx[0]=hx; sy[0]=hy;
      if(hx==fx&&hy==fy){ len++; fx=rnd()%32; fy=rnd()%24; if(fx<0)fx=0-fx; if(fy<0)fy=0-fy; }
    }
    cls();
    rect(fx*10,fy*10,9,9,0xFF5A5A);
    for(i=0;i<len;i++) rect(sx[i]*10,sy[i]*10,9,9, i==0?0xC8FF5A:0x2F9B5A);
    flip();
  }
  return 0; }
